cmake_minimum_required(VERSION 3.22)
project(study_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
if(ANDROID)
    # Inference must remain optimized in debuggable Android builds.
    add_compile_options(-O3)
    set(GGML_METAL OFF CACHE BOOL "" FORCE)
    set(GGML_BLAS OFF CACHE BOOL "" FORCE)
endif()
if(NOT WHISPER_SOURCE)
    set(WHISPER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../tools/local-stt/build/whisper.cpp")
endif()
if(NOT EXISTS "${WHISPER_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Run bash native-whisper/setup.sh before building this app")
endif()
add_subdirectory("${WHISPER_SOURCE}" whisper)
if(ANDROID)
    add_library(studywhisper SHARED study_whisper.cpp android_jni.cpp)
else()
    add_library(studywhisper STATIC study_whisper.cpp)
endif()
target_include_directories(studywhisper PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(studywhisper PRIVATE whisper)
if(ANDROID)
    target_link_options(studywhisper PRIVATE "-Wl,-z,max-page-size=16384")
endif()
